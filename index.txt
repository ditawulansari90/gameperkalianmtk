<script>
  // =========================================================
  // KODE RAHASIA EMAILJS BU DITA
  // =========================================================
  const PUBLIC_KEY = "0icbHDWAtrbQxSeEk"; // Tinggal ganti ini saja!
  const SERVICE_ID = "service_6b85k7p";
  const TEMPLATE_ID = "zz3caic";

  emailjs.init(PUBLIC_KEY); 

  const TEACHER_EMAIL = "ditawulansari90@gmail.com";
  let questions = [];
  let currentQ = 0;
  let score = 0;
  let timer;
  let timeLeft = 35;
  let studentName = "";
  let userAnswers = [];

  function generateQuestions() {
    let list = [];
    for (let i = 0; i < 10; i++) {
      let num1 = Math.floor(Math.random() * 3) + 6;
      let num2 = Math.floor(Math.random() * 10) + 1;
      list.push({ n1: num1, n2: num2, ans: num1 * num2 });
    }
    return list;
  }

  function startGame() {
    studentName = document.getElementById('student-name').value;
    if (!studentName.trim()) {
      alert("Harap masukkan nama terlebih dahulu!");
      return;
    }
    questions = generateQuestions();
    document.getElementById('start-screen').classList.add('hidden');
    document.getElementById('quiz-screen').classList.remove('hidden');
    showQuestion();
  }

  function showQuestion() {
    if (currentQ >= 10) {
      endGame();
      return;
    }
    document.getElementById('progress').innerText = `Soal ${currentQ + 1} dari 10`;
    document.getElementById('question-text').innerText = `${questions[currentQ].n1} x ${questions[currentQ].n2}`;
    document.getElementById('answer-input').value = '';
    
    clearInterval(timer);
    timeLeft = 35;
    document.getElementById('time-left').innerText = timeLeft;
    
    timer = setInterval(() => {
      timeLeft--;
      document.getElementById('time-left').innerText = timeLeft;
      if (timeLeft <= 0) {
        clearInterval(timer);
        recordAnswer(null);
      }
    }, 1000);
  }

  function submitAnswer() {
    let val = document.getElementById('answer-input').value;
    if (val === "") return;
    clearInterval(timer);
    recordAnswer(parseInt(val));
  }

  function recordAnswer(ans) {
    let correct = questions[currentQ].ans;
    let isCorrect = (ans === correct);
    if (isCorrect) score += 10;

    userAnswers.push({
      soal: `${questions[currentQ].n1} x ${questions[currentQ].n2}`,
      jawabanSiswa: ans !== null ? ans : "Waktu Habis",
      jawabanBenar: correct,
      status: isCorrect ? "Benar" : "Salah"
    });

    currentQ++;
    showQuestion();
  }

  function endGame() {
    document.getElementById('quiz-screen').classList.add('hidden');
    document.getElementById('result-screen').classList.remove('hidden');
    document.getElementById('res-name').innerText = studentName;
    document.getElementById('res-score').innerText = score;
  }

  function sendToTeacher() {
    const btn = document.getElementById('btn-send');
    const statusMsg = document.getElementById('status-msg');
    btn.disabled = true;
    statusMsg.innerText = "Sedang mengirim laporan ke email guru...";

    let ringkasanJawaban = "";
    userAnswers.forEach((item, index) => {
      ringkasanJawaban += `${index + 1}. ${item.soal} | Jwb Siswa: ${item.jawabanSiswa} | Jawaban Benar: ${item.jawabanBenar} (${item.status})\n`;
    });

    const templateParams = {
      to_email: TEACHER_EMAIL,
      student_name: studentName,
      score: score,
      quiz_details: ringkasanJawaban
    };

    emailjs.send(SERVICE_ID, TEMPLATE_ID, templateParams)
      .then(function() {
         statusMsg.style.color = "green";
         statusMsg.innerText = `Berhasil! Laporan kuis telah dikirim ke ${TEACHER_EMAIL}`;
      }, function(error) {
         statusMsg.style.color = "red";
         statusMsg.innerText = "Gagal mengirim. Periksa Public Key atau koneksi internet Anda.";
         btn.disabled = false;
      });
  }
</script>
